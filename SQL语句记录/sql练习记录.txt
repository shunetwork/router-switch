1、房源评分表练习case when 语句

select
count(case when 分数 <5 then 房源号 end) as '0-5',
count(case when 分数 >=5 and 分数 <7 then 房源号 end) as '5-7',
count(case when 分数 >=7 and 分数 <9 then 房源号 end) as '7-9',
count(case when 分数 >=9 then 房源号 end) as '9+'
from 评分表；

6、查询所有重复的学生姓名

select 姓名
from (
	select 姓名, count(姓名) as 计数
	from 学生表
	group by 姓名
) as 辅助表
where 计数 > 1;

select 姓名 from 学生表 group by 姓名 having count(姓名) > 1;
